GameManager.cs · instructor reference
using TMPro;
using UnityEngine;
using UnityEngine.SceneManagement;

public enum GameState
{
    Ready,
    Playing,
    Won,
    Lost
}

public class GameManager : MonoBehaviour
{
    [Header("Rules")]
    [SerializeField, Min(1)] private int targetScore = 3;
    [SerializeField, Min(1f)] private float roundDuration = 30f;

    [Header("UI")]
    [SerializeField] private TMP_Text scoreText;
    [SerializeField] private TMP_Text timerText;
    [SerializeField] private TMP_Text resultText;
    [SerializeField] private GameObject startPanel;
    [SerializeField] private GameObject resultPanel;

    [Header("Audio")]
    [SerializeField] private AudioSource sfxSource;
    [SerializeField] private AudioClip collectClip;
    [SerializeField] private AudioClip winClip;
    [SerializeField] private AudioClip loseClip;

    public GameState CurrentState { get; private set; }

    private int score;
    private float timeRemaining;

    private void Start()
    {
        ShowReady();
    }

    private void Update()
    {
        if (CurrentState != GameState.Playing)
        {
            return;
        }

        timeRemaining = Mathf.Max(
            0f,
            timeRemaining - Time.deltaTime
        );
        UpdateHud();

        if (timeRemaining <= 0f)
        {
            FinishRound(GameState.Lost);
        }
    }

    private void ShowReady()
    {
        CurrentState = GameState.Ready;
        score = 0;
        timeRemaining = roundDuration;
        Time.timeScale = 0f;

        startPanel.SetActive(true);
        resultPanel.SetActive(false);
        UpdateHud();
    }

    public void StartRound()
    {
        if (CurrentState != GameState.Ready)
        {
            return;
        }

        CurrentState = GameState.Playing;
        Time.timeScale = 1f;

        startPanel.SetActive(false);
        resultPanel.SetActive(false);
        UpdateHud();
    }

    public void AddScore(int amount)
    {
        if (CurrentState != GameState.Playing)
        {
            return;
        }

        score = Mathf.Min(score + amount, targetScore);
        PlaySfx(collectClip);
        UpdateHud();

        if (score >= targetScore)
        {
            FinishRound(GameState.Won);
        }
    }

    private void FinishRound(GameState result)
    {
        if (CurrentState != GameState.Playing)
        {
            return;
        }

        CurrentState = result;
        Time.timeScale = 0f;
        resultPanel.SetActive(true);

        bool didWin = result == GameState.Won;
        resultText.text = didWin ? "성공" : "시간 종료";
        PlaySfx(didWin ? winClip : loseClip);
    }

    public void RestartScene()
    {
        Time.timeScale = 1f;
        int currentScene = SceneManager.GetActiveScene().buildIndex;
        SceneManager.LoadScene(currentScene);
    }

    private void UpdateHud()
    {
        scoreText.text = $"{score} / {targetScore}";
        timerText.text = $"{timeRemaining:0.0}";
    }

    private void PlaySfx(AudioClip clip)
    {
        if (clip != null)
        {
            sfxSource.PlayOneShot(clip);
        }
    }
}

CurrentState != Playing

종료 뒤 추가 수집과 중복 종료를 막는 가장 앞쪽의 문입니다.

Mathf.Max(0f, ...)

남은 시간이 음수로 표시되지 않게 0에서 고정합니다.

Time.timeScale

Ready와 결과 상태에서 게임 시간을 0으로 멈추고 Start에서 1로 복원합니다.

UpdateHud()

점수나 시간이 바뀐 직후 같은 함수로 화면 표현을 다시 씁니다.

SceneManager.LoadScene

현재 Scene을 다시 불러와 수집물과 모든 실행 데이터를 초기 상태로 돌립니다.

중요한 주의

Time.timeScale = 0은 Update 호출 자체를 멈추는 것이 아니라 게임 시간의 진행을 멈춥니다. UI Button은 계속 클릭할 수 있습니다. Restart 전에 반드시 1로 복원해 다음 Scene이 멈춘 상태로 시작하지 않게 합니다.

수집 사건

수집물은 사건을 한 번만 전달합니다

접촉 감지와 전체 게임 규칙을 분리합니다. Collectible은 올바른 Player가 닿았는지 확인하고 GameManager에 점수 사건을 전달한 뒤 자신을 숨깁니다.

Collectible.cs · event sender
using UnityEngine;

public class Collectible : MonoBehaviour
{
    [SerializeField] private GameManager gameManager;

    private bool collected;

    private void OnTriggerEnter2D(Collider2D other)
    {
        if (collected || !other.CompareTag("Player"))
        {
            return;
        }

        if (gameManager.CurrentState != GameState.Playing)
        {
            return;
        }

        collected = true;
        gameManager.AddScore(1);
        gameObject.SetActive(false);
    }
}

검사통과할 때통과하지 못할 때
collected아직 처리하지 않은 수집물같은 사건의 중복 점수를 막고 return
CompareTag("Player")Player가 접촉다른 Collider 접촉은 무시
CurrentStatePlaying에서만 점수 허용Ready, Won, Lost에서는 무시
정확히 한 번 collected = true를 먼저 기록하고 점수를 전달합니다. 결과가 Won으로 바뀌어도 현재 수집 사건의 후속 처리는 예측 가능한 순서로 끝납니다.

오디오 피드백

소리는 사건의 차이를 즉시 알려 줍니다

효과음은 장면을 꾸미는 배경이 아니라 입력과 결과 사이의 짧은 확인 신호입니다. 세 소리는 길이와 역할을 분명히 다르게 합니다.

COLLECT

짧고 가벼운 확인

수집 직후 한 번 들립니다. 반복해서 들어도 피곤하지 않게 가장 짧고 작은 음량으로 시작합니다.

WIN

올라가는 완료 신호

목표 달성 뒤 한 번만 들립니다. Collect보다 길고 분명하지만 결과 Text를 가리지 않습니다.

LOSE

낮고 짧은 종료 신호

시간 종료 뒤 한 번만 들립니다. 실패를 알리되 불쾌하게 크거나 길지 않게 조절합니다.

MIX

작은 음량에서 시작

AudioSource Volume과 Clip 자체 음량을 구분합니다. 교실 스피커 기준으로 작게 시작해 필요한 만큼만 올립니다.

Play()

AudioSource의 기본 Clip 재생

AudioSource의 Clip 필드에 지정된 소리를 재생합니다. 같은 Source에서 다시 Play하면 재생 흐름이 바뀔 수 있습니다.

PlayOneShot(clip)

전달한 Clip을 한 번 재생

짧은 효과음에 적합합니다. 이미 PlayOneShot으로 재생 중인 Clip을 취소하지 않아 연속 수집음도 겹쳐 들릴 수 있습니다.

예측과 설명

코드보다 먼저 실행 결과를 말합니다

각 질문은 교수자 시연 직전에 사용합니다. 학생의 예측을 들은 뒤 한 줄만 바꾸거나 참조 하나만 빼서 결과를 확인합니다.

Q1. Result Panel 참조가 None이면 언제 오류가 납니까?

ShowReady에서 resultPanel.SetActive(false)를 호출하므로 Scene 시작 직후 오류가 납니다. 오류가 성공 순간에만 생긴다고 단정하면 호출 순서를 놓친 것입니다.

Q2. Won 뒤 수집물에 닿으면 점수가 오릅니까?

오르지 않습니다. Collectible과 AddScore가 모두 Playing인지 검사하고, Won에서는 곧바로 return합니다.

Q3. RestartScene 전에 Time Scale을 1로 돌리는 이유는?

Time.timeScale은 Scene을 다시 불러와도 전역 값으로 남을 수 있습니다. 먼저 1로 복원해야 새 Scene의 초기화가 멈춘 시간 위에서 시작하지 않습니다.

Q4. Editor에서는 재시작되는데 빌드에서 실패한다면?
